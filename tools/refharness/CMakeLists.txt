cmake_minimum_required(VERSION 3.20)
project(js8refharness CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)
list(APPEND CMAKE_PREFIX_PATH /opt/homebrew/opt/qt@5 /opt/homebrew)
find_package(Qt5 REQUIRED COMPONENTS Core)
find_package(Boost REQUIRED)
set(REF ${CMAKE_SOURCE_DIR}/../../reference/js8call)
add_executable(js8ref src/main.cpp src/JS8_ref.cpp src/varicode.cpp src/jsc.cpp src/jsc_map.cpp src/jsc_list.cpp src/decodedtext.cpp)
target_compile_definitions(js8ref PRIVATE FFTW_ESTIMATE_PATIENT=FFTW_ESTIMATE)
target_include_directories(js8ref PRIVATE src ${REF} ${REF}/vendor /opt/homebrew/include ${Boost_INCLUDE_DIRS})
target_link_directories(js8ref PRIVATE /opt/homebrew/lib)
target_link_libraries(js8ref Qt5::Core fftw3f)
add_executable(jscdump src/jscdump.cpp src/jsc_map.cpp src/jsc_list.cpp)
target_include_directories(jscdump PRIVATE src ${REF})
target_link_libraries(jscdump Qt5::Core)
